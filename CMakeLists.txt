cmake_minimum_required(VERSION 3.24)
project(ndi_mxl_ingest VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(INGEST_WITH_NDI "Build with externally installed NDI SDK" OFF)
option(INGEST_WITH_MXL "Build with installed MXL 1.1" OFF)
option(INGEST_SANITIZE "Enable ASan and UBSan" OFF)
set(NDI_ROOT "" CACHE PATH "NDI SDK root")
find_package(Threads REQUIRED)
add_library(ingest_core src/core.cpp src/config.cpp)
target_include_directories(ingest_core PUBLIC include)
target_compile_options(ingest_core PUBLIC -Wall -Wextra -Wconversion -Wshadow -Werror)
target_link_libraries(ingest_core PUBLIC Threads::Threads)
if(INGEST_SANITIZE)
  target_compile_options(ingest_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(ingest_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(ndi-mxl-ingest src/main.cpp src/ndi.cpp src/mxl.cpp)
target_link_libraries(ndi-mxl-ingest PRIVATE ingest_core)
if(INGEST_WITH_NDI)
  find_path(NDI_INCLUDE_DIR Processing.NDI.Lib.h HINTS ${NDI_ROOT}/include REQUIRED)
  find_library(NDI_LIBRARY ndi HINTS ${NDI_ROOT}/lib ${NDI_ROOT}/lib/x86_64-linux-gnu ${NDI_ROOT}/lib/aarch64-linux-gnu REQUIRED)
  target_include_directories(ndi-mxl-ingest SYSTEM PRIVATE ${NDI_INCLUDE_DIR})
  target_link_libraries(ndi-mxl-ingest PRIVATE ${NDI_LIBRARY})
  target_compile_definitions(ndi-mxl-ingest PRIVATE INGEST_WITH_NDI)
endif()
if(INGEST_WITH_MXL)
  find_path(MXL_INCLUDE_DIR mxl/flow.h REQUIRED)
  find_library(MXL_LIBRARY mxl REQUIRED)
  target_include_directories(ndi-mxl-ingest SYSTEM PRIVATE ${MXL_INCLUDE_DIR})
  target_link_libraries(ndi-mxl-ingest PRIVATE ${MXL_LIBRARY})
  target_compile_definitions(ndi-mxl-ingest PRIVATE INGEST_WITH_MXL)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(ingest_tests tests/tests.cpp)
  target_link_libraries(ingest_tests PRIVATE ingest_core)
  add_test(NAME core COMMAND ingest_tests)
  add_test(NAME simulation COMMAND ndi-mxl-ingest --simulate --frames 3)
  add_test(NAME config COMMAND ndi-mxl-ingest --config ${CMAKE_CURRENT_SOURCE_DIR}/config/ingest.conf --simulate --frames 2)
  add_test(NAME invalid_cli COMMAND ndi-mxl-ingest --frames -1)
  set_tests_properties(invalid_cli PROPERTIES WILL_FAIL TRUE)
endif()
install(TARGETS ndi-mxl-ingest RUNTIME DESTINATION bin)
